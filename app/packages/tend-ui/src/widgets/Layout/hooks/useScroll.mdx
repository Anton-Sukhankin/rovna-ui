# useScroll

## Использование

```tsx
import { useScroll } from '@rovna-ui/components/widgets';

const ComponentOne = () => {
  const ref = React.useRef(null);
  const { register, scroll } = useScroll();
  register('ComponentOne', ref);

  return (
    <div
      ref={ref}
      onClick={() => {
        scroll('ComponentTwo');
      }}
    >
      Компонент 1
    </div>
  );
};

const ComponentTwo = () => {
  const ref = React.useRef(null);
  const { register, scroll } = useScroll();
  register('ComponentTwo', ref);

  return (
    <div
      ref={ref}
      onClick={() => {
        scroll('ComponentOne');
      }}
    >
      Компонент 1
    </div>
  );
};
```

## Интерфейсы

```ts
type UseScrollReturn = {
  /**
   * Функция для скроллинга до зарегистрированного элемента
   * @param key - уникальное имя ноды
   */
  scroll: (key: string) => void;
  /**
   * Функция регистрации компонента
   */
  register: (key: string, ref: React.RefMutableObject<T>) => void;
};
```

> Важно! `header` - зарезерированный ключ. Вызов `scroll("header")` прокрутит страницу до шапки
