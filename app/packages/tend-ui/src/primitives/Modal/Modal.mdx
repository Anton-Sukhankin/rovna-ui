# Modal

## Использование

```tsx
import { Modal } from '@rovna-ui/components/primitives';
```

### Для лучшего `tree-shaking`

```tsx
import { Modal } from '@rovna-ui/components/primitives/Modal';
```

### Usage

```tsx
const Page = () => {
  const [open, setOpen] = React.useState(false);

  return (
    <>
      <Button
        onClick={() => {
          setOpen(true);
        }}
      >
        Open
      </Button>
      <Modal
        open={open}
        onCancel={() => {
          setOpen(false);
        }}
      />
    </>
  );
};
```
