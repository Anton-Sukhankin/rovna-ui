# Цветовая палитра глобальных продуктов

## Использование

```tsx
import { colors } from '@rovna-ui/tokens/global';
import { useColors } from '@rovna-ui/components/theme';
```

## Для лучшего `tree-shaking`

```tsx
import { colors } from '@rovna-ui/tokens/global/colors';
import { useColors } from '@rovna-ui/components/theme';
```

## Палитра

<ColorPalette>
  {Object.entries(colors).map(([key, value]) => (
    <ColorItem title={key} colors={{ [key]: value }} />
  ))}
</ColorPalette>
