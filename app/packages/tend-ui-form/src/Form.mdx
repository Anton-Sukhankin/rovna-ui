# useForm

<Meta of={FormStories} title='Rovna UI/Form' />

## Использование

### Built-in

```tsx
import { useForm, Form } from '@rovna-ui/form';
import { Input, TextArea } from '@rovna-ui/components/primitives';

const Component = () => {
  const form = useForm<{ name?: string; surname?: string; comment?: string }>({
    onChange: (state) => {},
    onSubmit: (state) => {},
  });

  return (
    <Form form={form}>
      {/* Можно передавать дженерики для выведения типов */}
      {/* Первый аргумент - значение из onChange компонента children */}
      {/* Второй аргумент - получаемое значение */}
      <Form.Field<React.KeyboardEvent, string>
        name='name'
        {/* Тут значения выведутся автоматически */}
        getValueFromEvent={e => e.target.value}
      >
        <Input />
      </Form.Field>
      <Form.Field<React.KeyboardEvent, string>
        name='surname'
        getValueFromEvent={e => e.target.value}
      >
        <Input />
      </Form.Field>
      <Form.Field<React.KeyboardEvent, string>
        name='comment'
        getValueFromEvent={e => e.target.value}
      >
        <TextArea />
      </Form.Field>
    </Form>
  );
};
```
