# Filters

## Использование

```tsx
import { Filters } from '@rovna-ui/filters';
```

## Интерфейсы

### FilterConfig

Схема фильтра

```tsx
type FilterConfig = {
  /**
   * `React.key`
   */
  key?: React.Key;
  /**
   * Уникальный идентификатор фильтра
   */
  id: string;
  /**
   * Имя фильтра
   */
  name: string;
  /**
   * Лейбл в списке
   */
  label?: React.ReactNode;
  /**
   * Тип фильтра
   * `Input`, `Select`, `Search`, `Checkbox`, `Radio`
   */
  component: ComponentPickerProps;
  /**
   * Массив зависимостей.
   * Если один фильтр должен недоступен пока не выбран другой,
   * то используйте это свойство
   */
  depends?: string[];
};
```

#### Пример

```tsx
const filters: FilterConfig[] = [
  {
    id: 'provider',
    name: 'provider',
    label: 'Поставщик',
    component: {
      component: 'select',
      options: [
        { value: '1', label: 'ООО Интерстрой' },
        { value: '2', label: 'ООО Радуга' },
        { value: '3', label: 'ООО СтройМонтаж' },
      ],
    },
  },
];
```
