# useUpload

## Использование

```tsx
import { useUpload } from '@rovna-ui/upload';

const Upload = () => {
  const upload = useUpload();

  return (
    <div>
      <button onClick={upload.getOpenHandler()}>Открыть инспектор</button>
      <input {...upload.getInputProps()} />
    </div>
  );
};
```
