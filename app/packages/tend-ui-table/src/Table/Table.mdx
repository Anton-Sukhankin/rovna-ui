# Table

## Использование

### Вариант 1 (предпочтительно)

```tsx
import React from 'react';
import { useColumns, DrawerColumnsSettings } from '@rovna-ui/columns-settings';
import { Table } from '@rovna-ui/table';

const _columns: Table.ColumnType[] = [
  {
    id: 'contractor',
    label: 'Подрядчик',
    title: 'Подрядчик',
    onHeaderCell: () => ({
      id: 'contractor',
    }),
  },
];

const filters: Table.FilterConfig[] = [
  {
    key: 'filter-contractor',
    id: 'contractor',
    name: 'contractor',
    label: 'Подрядчик',
    component: { component: 'input', placeholder: 'Введите имя подрядчика' },
  },
  {
    key: 'filter-provider',
    id: 'provider',
    name: 'provider',
    label: 'Поставщик',
    component: { component: 'checkbox', placeholder: 'Выберите поставщика' },
  },
];

const sorters: Table.SorterConfig[] = [
  {
    key: 'sorter-contractor',
    id: 'contractor',
    name: 'contractor',
  },
];

type TableFilters = {
  contractor: string;
  provider: string[];
};

const settings = useColumns({ defaultColumns: _columns });

<Table.Root<TableFilters>
  settings={settings}
  value={{
    filters: {
      contractor: 1,
      provider: [2]
    },
    sorters: {
      contractor: 'ascend',
      provider: 'descend'
    },
    search: 'Hello World'
  }}
  filters={filters}
  sorters={sorters}
  onFilterValuesChange={(changed, values) => {}}
  onSorterValuesChange={(changed, values) => {}}
  onSearchValueChange={(changed value) => {}}
>
  <Table.Header>
    <Table.ControlPanel />
    <Table.Search />
    <Table.HotFilters />
  </Table.Header>
  <Table.Table
    columns={settings.getAntdTableColumns()}
    components={{
      header: {
        cell: (props: React.HTMLAttributes<HTMLTableCellElement>) => (
          <Table.ContextMenu id={props.id}>
            <Table.HeaderCell {...props} />
          </Table.ContextMenu>
        ),
      },
    }}
  />
  <Table.Filters />
  <DrawerColumnsSettings settings={settings} />
</Table.Root>;
```
