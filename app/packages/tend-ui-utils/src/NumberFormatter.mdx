# NumberFormatter

## Использование

```tsx
import { NumberFormatter } from '@rovna-ui/utils';
```

### Для лучшего `tree-shaking`

```tsx
import { NumberFormatter } from '@rovna-ui/utils/NumberFormatter';
```

## Интерфейсы

```tsx
type NumberFormatterOptions = {
  currency?: true;
  minimumFractionDigits?: number;
  maximumFractionDigits?: number;
};
```

```js
NumberFormatter.format(1); // 1
NumberFormatter.format(10); // 10
NumberFormatter.format(100); // 100
NumberFormatter.format(1000); // 1 000
NumberFormatter.format(10000); // 1 000

NumberFormatter.format(1.5); // 1,5
NumberFormatter.format(10.5); // 10,5
NumberFormatter.format(100.5); // 100,5
NumberFormatter.format(1000.5); // 1 000,5
NumberFormatter.format(10000.5); // 1 000,5

NumberFormatter.format(1.5, { minimumFractionDigits: 3 }); // 1,500
NumberFormatter.format(10.5, { minimumFractionDigits: 3 }); // 10,500
NumberFormatter.format(100.5, { minimumFractionDigits: 3 }); // 100,500
NumberFormatter.format(1000.5, { minimumFractionDigits: 3 }); // 1 000,500
NumberFormatter.format(10000.5, { minimumFractionDigits: 3 }); // 1 000,500

NumberFormatter.format(1, { currency: true }); // 1,00 ₽
```
