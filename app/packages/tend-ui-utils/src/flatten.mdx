# flatten

## Использование

```tsx
import { flatten } from '@rovna-ui/utils';
```

### Для лучшего `tree-shaking`

```tsx
import { flatten } from '@rovna-ui/utils/flatten';
```

```js
console.log(
  flatten([
    {
      key: '1',
      value: 'Node 1',
      children: [
        {
          key: '2',
          value: 'Node 2',
          children: [
            {
              key: '3',
              value: 'Node 3',
              children: [
                { key: '4', value: 'Node 4', children: [{ key: '5', value: 'Node 5' }] },
              ],
            },
          ],
        },
      ],
    },
  ]),
);
/*
[
  { key: '1', value: 'Node 1' },
  { key: '2', value: 'Node 2' },
  { key: '3', value: 'Node 3' },
  { key: '4', value: 'Node 4' },
  { key: '5', value: 'Node 5' },
]
*/
```
