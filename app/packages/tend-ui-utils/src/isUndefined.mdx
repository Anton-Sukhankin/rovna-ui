# isUndefined

## Использование

```tsx
import { isUndefined } from '@rovna-ui/utils';
```

### Для лучшего `tree-shaking`

```tsx
import { isUndefined } from '@rovna-ui/utils/isUndefined';
```

```js
console.log(isUndefined(1)); // false
console.log(isUndefined('Hello World')); // false
console.log(isUndefined(() => {})); // false
console.log(isUndefined({})); // false
console.log(isUndefined([])); // false
console.log(isUndefined(null)); // false
console.log(isUndefined(true)); // false
console.log(isUndefined(false)); // false

console.log(isUndefined(undefined)); // true
```
